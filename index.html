// anga/js/weather.js
const CACHE_TTL = 15 * 60 * 1000;
const keys = loadKeys();

async function fetchWeather(lat, lon) {
  const hit = await cache.get(key(lat, lon));
  if (hit && !isStale(hit)) return hit;

  const apiKey = keys.pick(usage);
  const url = `${API}?lat=${lat}&lon=${lon}&key=${apiKey}`;
  const res = await fetch(url);
  if (!res.ok) {
    if (res.status === 429) rotateKey();
    throw new WeatherError(res.status);
  }

  const data = await res.json();
  await cache.set(key(lat, lon), data, CACHE_TTL);
  usage.increment(apiKey);
  return data;
}

function pickKey(usage) {
  return keys
    .filter(k => usage.get(k) < QUOTA)
    .sort((a, b) => usage.get(a) - usage.get(b))[0];
}

export { fetchWeather, pickKey };

# studybuddy/tutor/views.py
from django.views.decorators.http import require_POST
from .retriever import retrieve_passages
from .llm import call_llm, verify_citations

@require_POST
@login_required
def ask_tutor(request):
    q = request.POST.get("question", "").strip()
    if not q:
        return JsonResponse(
            {"error": "empty_question"}, status=400
        )

    notes = Note.objects.filter(owner=request.user)
    passages = retrieve_passages(q, notes, k=4)

    answer, cites = call_llm(q, passages)
    if not verify_citations(answer, cites):
        return JsonResponse({
            "answer": "I could not find that in your notes.",
            "citations": [],
            "confidence": "low",
        })

    return JsonResponse({
        "answer": answer,
        "citations": [c.id for c in cites],
        "confidence": "high",
    })

    summary: "toggle color scheme",

      const root = document.documentElement;
      root.setAttribute("data-theme", next);
      try {
        localStorage.setItem("melvin.theme", next);
      } catch (e) {}

  if (!handler) return `command not found: ${name}`;
  return handler.run(args);

Anga — Weather for Kenya

A weather dashboard engineered to survive a 1,000-requests-per-month API quota.

The upstream provider allows roughly 33 calls a day, across every location. Anga puts a Django service between users and the API: Redis caching with honest freshness labels, request coalescing so concurrent misses cost one upstream call, a circuit breaker for quota lockouts, and location search served from its own PostgreSQL gazetteer.

▸ Verified live: 30 simultaneous requests → exactly 1 upstream call. Request coalescing protects the quota under spiky traffic.

▸ 222 automated tests. 98 backend, 124 frontend. Concurrency, mocked outages, accessibility contrast.

StudyBuddy

An AI study tool that summarizes only from the student’s own notes, and verifies that it did.

Students upload notes as PDF, Word, text or a lecture recording and get summaries, quizzes and flashcards grounded strictly in their material. Grounding is enforced after generation: definitions must overlap the source and every quiz question must quote an evidence sentence. Custom JWT auth, file-signature upload checks and a quota-aware fallback chain across Gemini models.

▸ ~31 API endpoints and 14,600 lines of hand-written code across a Django API and React 19 frontend.

~/about

About me

I’m a software developer based in Kenya. I build web applications end to end: the API, the data layer, the frontend and the deployment behind them. The parts of engineering most people skip are the ones I enjoy most: what happens when the quota runs out, the upstream API lies about its headers, or the user is on a slow connection.

My habit is to verify rather than assume. For Anga I probed the provider’s live API and found its documented rate-limit headers didn’t exist. That discovery shaped the whole architecture. For StudyBuddy I didn’t trust the AI’s promises to stay grounded, so the code checks every definition and quiz question against the source notes after generation. If a reading is stale, the interface says so instead of pretending.

I’m interested in software development roles, internships and freelance work: building products for companies, and websites and SEO for businesses that need to be found. If that sounds like something you’re working on, get in touch.
